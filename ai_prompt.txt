# 6️⃣ Middleware (Next 13)
# ----------------------------------------------------------------
Créer **`middleware.js`** :
```js
import { supabase } from './utils/supabaseClient';
import { NextResponse } from 'next/server';
export async function middleware(req) {
  const token = req.cookies['sb:token'] ?? '';
  const { data: { user } } = await supabase.auth.getUser(token);
  if (!user) return NextResponse.redirect(new URL('/login', req.url));
  return NextResponse.next();
}
export const config = { matcher: ['/student/:path*', '/teacher/:path*'] };
```

# 9️⃣ API d’historique des leçons
# ----------------------------------------------------------------
Créer `pages/api/lessons.js` :
```js
export default async function handler(req, res) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  const { data: { user } } = await supabase.auth.getUser(token);
  if (!user) return res.status(401).json({ error: 'Unauthorized' });
  const { data, error } = await supabase
    .from('lessons')
    .select('id, instruction, pdf_path, created_at')
    .eq('student_id', user.id)
    .order('created_at', { ascending: false });
  if (error) return res.status(500).json({ error });
  const publicUrls = data.map(l => ({
    ...l,
    pdfUrl: supabase.storage.from('lesson-pdfs').getPublicUrl(l.pdf_path).publicUrl
  }));
  res.status(200).json(publicUrls);
}
```
